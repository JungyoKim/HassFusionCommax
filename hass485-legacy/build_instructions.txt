# Go 빌드 환경별 명령어 모음

## 1. 윈도우용 실행파일 빌드

### PowerShell
$env:GOOS="windows"
$env:GOARCH="amd64"
go build -o usbrs485.exe

### cmd(명령 프롬프트)
set GOOS=windows
set GOARCH=amd64
go build -o usbrs485.exe

---

## 2. 리눅스(라즈베리파이/ARM)용 실행파일 빌드

### PowerShell
$env:GOOS="linux"
$env:GOARCH="arm"
$env:GOARM="7"
go build -o hass485

### cmd(명령 프롬프트)
set GOOS=linux
set GOARCH=arm
set GOARM=7
go build -o usbrs485

---

## 3. 리눅스(x86_64 서버)용 실행파일 빌드

### PowerShell
$env:GOOS="linux"
$env:GOARCH="amd64"
go build -o usbrs485

### cmd(명령 프롬프트)

---

## 4. 맥(MacOS, Intel/Apple Silicon)용 실행파일 빌드

### Intel 맥 (x86_64)
$env:GOOS="darwin"
$env:GOARCH="amd64"
go build -o usbrs485

set GOOS=darwin
set GOARCH=amd64
go build -o usbrs485

### Apple Silicon 맥 (M1/M2, arm64)
$env:GOOS="darwin"
$env:GOARCH="arm64"
go build -o usbrs485

set GOOS=darwin
set GOARCH=arm64
go build -o usbrs485

---

## 5. 환경변수 리셋(윈도우)

### PowerShell
Remove-Item Env:GOOS
Remove-Item Env:GOARCH
Remove-Item Env:GOARM

go build -o usbrs485.exe

### cmd(명령 프롬프트)
set GOOS=
set GOARCH=
set GOARM=

go build -o usbrs485.exe
